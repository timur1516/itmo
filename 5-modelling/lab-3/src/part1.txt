****************************************************************************
*                              Модель СМО G/G/K/E                          *
****************************************************************************
*                        И с х о д н ы е   д а н н ы е                     *
****************************************************************************
b_avg       EQU         15          ; средняя длительность обслуживания
lambda      EQU         0.3         ; интенсивность входного потока

E_buf_1     EQU         2           ; емкость накопителя (буфера)

t_a         EQU         1/lambda    ; средний интервал между поступающими заявками                        
t_b         EQU         b_avg	    ; средняя длительность обслуживания заявки в приборе

RN_a        EQU         20		    ; номер генератора для потока                                          
RN_b	    EQU         553		    ; номер генератора для длительности обслуживания
****************************************************************************
NODE_1	    STORAGE	    1           ; прибор 1
NODE_2	    STORAGE	    1		    ; прибор 2
NODE_3	    STORAGE	    1		    ; прибор 3
****************************************************************************
TIME_ALL	TABLE	    M1,1,1,20		        ; время пребывания в системе
BUF_LEN_1   QTABLE	    BUF_1,0.5,0.5,10		; статистика по очереди 1
****************************************************************************
*                           Основной цикл симуляции                        *
****************************************************************************
GENERATE	(Exponential(RN_a,0,t_a))

TRANSFER 	0.5,,TO_N1
TRANSFER 	0.3,,TO_N2
TRANSFER	,TO_N3

TO_N1	TEST	L	Q$BUF_1,E_buf_1,Reject		; проверка места в буфере
	QUEUE	BUF_1						; размещение заявки в очереди
	ENTER	NODE_1						; захват прибора
	DEPART	BUF_1						; уход заявки из очереди
	ADVANCE	(Exponential(RN_b,0,t_b))				; обслуживание по экспоненциальному распределению
	LEAVE	NODE_1						; освобождение прибора
	TABULATE	TIME_ALL
	TERMINATE	1						; завершение транзакта

TO_N2	TEST	E	S$NODE_2,0,Reject		; проверка занятости прибора
	ENTER	NODE_2
	ADVANCE	(Exponential(RN_b,0,t_b))
	LEAVE	NODE_2
	TABULATE	TIME_ALL
	TERMINATE	1

TO_N3	TEST	E	S$NODE_3,0,Reject
	ENTER	NODE_3
	ADVANCE	(Exponential(RN_b,0,t_b))
	LEAVE	NODE_3
	TABULATE	TIME_ALL
	TERMINATE	1

Reject	TERMINATE

GENERATE	100
	SAVEVALUE	Util1,(FR$NODE_1/0.01)
	SAVEVALUE	Util2,(FR$NODE_2/0.01) 
	SAVEVALUE	Util3,(FR$NODE_3/0.01)
	SAVEVALUE	QueueLen,Q$BUF_1
	TERMINATE	1

START	10000
